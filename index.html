Google Sheets & Drive Sync

ซิงก์ข้อมูลรูปภาพและเสื้อผ้าลงในกูเกิลไดรฟ์ของคุณ

นำ URL ที่ได้จากการ Deploy Google Apps Script มาวางตรงนี้

วิธีสร้าง Google Apps Script (คลิกเพื่อดู)

  1. สร้าง Google Sheet ใหม่ แล้วไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. วางโค้ดสคริปต์ด้านล่างนี้แทนที่ทั้งหมด แล้วกดบันทึก
  3. กด Deploy > New Deployment เลือกประเภทเป็น Web app
  4. ตั้งค่า "Who has access" เป็น Anyone (ทุกคน) แล้วกด Deploy
  5. คัดลอก Web App URL มาใส่ในช่องด้านบน
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  if (data.action === "sync_items") {
    sheet.clear();
    sheet.appendRow(["ID", "Title", "Category", "Brand", "Color", "ImageURL", "Favorite"]);
    data.items.forEach(function(item) {
      sheet.appendRow([item.id, item.title, item.category, item.brand, item.color, item.image, item.favorite]);
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  }
}
