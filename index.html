// ========================================== // MY CLOSET BACKEND // Google Sheets + Google Drive // Version: CLEAN / SYNC ITEMS + OUTFITS + EDIT & DELETE // ========================================== const FOLDER_NAME = "My Closet Images"; const OUTFIT_SHEET_NAME = "OUTFITS"; const ITEM_SHEET_NAME = "ITEMS"; // ========================================== // GET DATA (โหลดเสื้อผ้า + Outfits) // ========================================== function doGet(e) { try { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); // 1. LOAD ITEMS const itemSheet = getItemSheet(); const itemValues = itemSheet.getDataRange().getValues(); let items = []; if (itemValues.length > 1) { const headers = itemValues[0]; items = itemValues.slice(1) .filter(row => row.some(value => value !== "")) .map(row => { const item = {}; headers.forEach((header, index) => { item[header] = row[index]; }); return { id: String(item.ID || ""), title: item.Title || "", category: item.Category || "", brand: item.Brand || "", color: item.Color || "", image: item.ImageURL || "", favorite: item.Favorite === true || item.Favorite === "TRUE" }; }); } // 2. LOAD OUTFITS const outfitSheet = getOutfitSheet(); const outfitValues = outfitSheet.getDataRange().getValues(); let outfits = []; if (outfitValues.length > 1) { const headers = outfitValues[0]; outfits = outfitValues.slice(1) .filter(row => row.some(value => value !== "")) .map(row => { const outfit = {}; headers.forEach((header, index) => { outfit[header] = row[index]; }); let itemIds = []; try { if (outfit.ItemIds) itemIds = JSON.parse(outfit.ItemIds); } catch (error) { itemIds = []; } return { id: String(outfit.ID || ""), title: outfit.Title || outfit.Name || "", name: outfit.Name || outfit.Title || "", occasion: outfit.Occasion || "", itemIds: Array.isArray(itemIds) ? itemIds : [], createdAt: outfit.CreatedAt || "" }; }); } const responseData = JSON.stringify({ status: "success", items: items, outfits: outfits }); const callback = e && e.parameter ? e.parameter.callback : null; if (callback) { return ContentService.createTextOutput(callback + "(" + responseData + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(responseData) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return createResponse({ status: "error", message: error.toString() }); } } // ========================================== // POST DATA (จัดการ Action ต่างๆ) // ========================================== function doPost(e) { try { if (!e || !e.postData) { return createResponse({ status: "error", message: "ไม่มีข้อมูล POST" }); } const data = JSON.parse(e.postData.contents); if (data.action === "add_item") return addItem(data); if (data.action === "update_item") return updateItem(data); if (data.action === "delete_item") return deleteItem(data); if (data.action === "sync_items") return syncItems(data); if (data.action === "save_outfit") return saveOutfit(data); if (data.action === "sync_outfits") return syncOutfits(data); if (data.action === "delete_outfit") return deleteOutfit(data); return createResponse({ status: "error", message: "Unknown action: " + data.action }); } catch (error) { return createResponse({ status: "error", message: error.toString() }); } } // ========================================== // ADD ITEM // ========================================== function addItem(data) { const item = data.item || {}; const sheet = getItemSheet(); let imageUrl = ""; if (item.image && item.image.startsWith("data:image")) { const parts = item.image.split(","); if (parts.length >= 2) { const mimeMatch = parts[0].match(/:(.*?);/); const mimeType = mimeMatch ? mimeMatch[1] : "image/jpeg"; const base64Data = parts[1]; const blob = Utilities.newBlob( Utilities.base64Decode(base64Data), mimeType, String(item.title || "item") + "_" + String(item.id || Date.now()) ); const folder = getImageFolder(); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); imageUrl = "https://drive.google.com/uc?export=view&id=" + file.getId(); } } else if (item.image) { imageUrl = item.image; } sheet.appendRow([ String(item.id || Date.now()), item.title || "", item.category || "", item.brand || "", item.color || "", imageUrl, item.favorite || false, new Date() ]); return createResponse({ status: "success", message: "เพิ่มเสื้อผ้าเรียบร้อย", imageUrl: imageUrl }); } // ========================================== // UPDATE ITEM // ========================================== function updateItem(data) { const item = data.item || {}; const itemId = String(item.id || "").trim(); if (!itemId) return createResponse({ status: "error", message: "ไม่พบ Item ID" }); const sheet = getItemSheet(); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === itemId) { if (item.title !== undefined) sheet.getRange(i + 1, 2).setValue(item.title); if (item.category !== undefined) sheet.getRange(i + 1, 3).setValue(item.category); if (item.brand !== undefined) sheet.getRange(i + 1, 4).setValue(item.brand); if (item.color !== undefined) sheet.getRange(i + 1, 5).setValue(item.color); SpreadsheetApp.flush(); return createResponse({ status: "success", message: "แก้ไขข้อมูลสินค้าเรียบร้อย" }); } } return createResponse({ status: "error", message: "ไม่พบสินค้าที่ต้องการแก้ไข ID: " + itemId }); } // ========================================== // DELETE ITEM // ========================================== function deleteItem(data) { const itemId = String(data.id || "").trim(); if (!itemId) return createResponse({ status: "error", message: "ไม่พบ Item ID" }); const sheet = getItemSheet(); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === itemId) { sheet.deleteRow(i + 1); return createResponse({ status: "success", message: "ลบเสื้อผ้าสำเร็จ" }); } } return createResponse({ status: "error", message: "ไม่พบรายการที่ต้องการลบ" }); } // ========================================== // SYNC ITEMS // ========================================== function syncItems(data) { const items = Array.isArray(data.items) ? data.items : []; const sheet = getItemSheet(); sheet.clear(); sheet.appendRow(["ID", "Title", "Category", "Brand", "Color", "ImageURL", "Favorite", "CreatedAt"]); items.forEach(item => { sheet.appendRow([ String(item.id || Date.now()), item.title || "", item.category || "", item.brand || "", item.color || "", item.image || "", item.favorite || false, new Date() ]); }); return createResponse({ status: "success", message: "Sync Items สำเร็จ", count: items.length }); } // ========================================== // SAVE OUTFIT // ========================================== function saveOutfit(data) { const outfit = data.outfit || {}; const sheet = getOutfitSheet(); const id = String(outfit.id || Date.now()); const title = outfit.title || outfit.name || "My Outfit"; const occasion = outfit.occasion || ""; const itemIds = Array.isArray(outfit.itemIds) ? outfit.itemIds : []; sheet.appendRow([id, title, occasion, JSON.stringify(itemIds), new Date()]); return createResponse({ status: "success", message: "บันทึก Outfit สำเร็จ", outfit: { id: id, title: title, name: title, occasion: occasion, itemIds: itemIds } }); } // ========================================== // SYNC OUTFITS // ========================================== function syncOutfits(data) { const outfits = Array.isArray(data.outfits) ? data.outfits : []; const sheet = getOutfitSheet(); sheet.clear(); sheet.appendRow(["ID", "Title", "Occasion", "ItemIds", "CreatedAt"]); outfits.forEach(outfit => { sheet.appendRow([ String(outfit.id || Date.now()), outfit.title || outfit.name || "My Outfit", outfit.occasion || "", JSON.stringify(Array.isArray(outfit.itemIds) ? outfit.itemIds : []), new Date() ]); }); return createResponse({ status: "success", message: "Sync Outfits สำเร็จ", count: outfits.length }); } // ========================================== // DELETE OUTFIT // ========================================== function deleteOutfit(data) { const outfitId = String(data.id || "").trim(); if (!outfitId) return createResponse({ status: "error", message: "ไม่พบ Outfit ID" }); const sheet = getOutfitSheet(); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === outfitId) { sheet.deleteRow(i + 1); return createResponse({ status: "success", message: "ลบ Outfit สำเร็จ" }); } } return createResponse({ status: "error", message: "ไม่พบ Outfit ที่ต้องการลบ" }); } // ========================================== // HELPER FUNCTIONS // ========================================== function getItemSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const requiredHeaders = ["ID", "Title", "Category", "Brand", "Color", "ImageURL", "Favorite", "CreatedAt"]; const sheets = spreadsheet.getSheets(); for (let i = 0; i < sheets.length; i++) { const sheet = sheets[i]; if (sheet.getLastColumn() === 0) continue; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (requiredHeaders.every(h => headers.indexOf(h) !== -1)) return sheet; } let sheet = spreadsheet.getSheetByName(ITEM_SHEET_NAME) || spreadsheet.insertSheet(ITEM_SHEET_NAME); sheet.clear(); sheet.appendRow(requiredHeaders); return sheet; } function getOutfitSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); let sheet = spreadsheet.getSheetByName(OUTFIT_SHEET_NAME) || spreadsheet.insertSheet(OUTFIT_SHEET_NAME); const headers = ["ID", "Title", "Occasion", "ItemIds", "CreatedAt"]; if (sheet.getLastColumn() === 0) { sheet.appendRow(headers); } return sheet; } function getImageFolder() { const folders = DriveApp.getFoldersByName(FOLDER_NAME); return folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER_NAME); } function createResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
